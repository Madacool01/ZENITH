vector<int> / vector<long long>

            

              
Template

              
vector<int> <name> <size> <min> <max> [unique] [sort];

            

            

              
Parameters

              
    
                
  • name — variable name (letters, numbers, _)
    • 
                
  • size — number of elements to generate
    • 
                
  • min — minimum value (inclusive)
    • 
                
  • max — maximum value (inclusive)
    • 
                
  • true (optional 5th arg) — all values must be distinct
    • 
                
  • true (optional 6th arg) — sort in ascending order
    • 
              

            

            

              
Example

              
vector<int> a 10 1 100;

              
→ 10 random integers, each between 1 and 100.

            

          


          

            
string

            

              
Template

              
string <name> <length> <charset> [unique];

            

            

              
Parameters

              
    
                
  • name — variable name
    • 
                
  • length — number of characters
    • 
                
  • charset — binary | lowercase | uppercase | digits | alpha | alphanum | custom{...} for an exact set
    • 
                
  • true (optional) — all characters unique
    • 
              

            

            

              
Example

              
string s 10 lowercase;

              
→ 10 random lowercase letters.

              
string s 6 custom{Ab01};

              
→ 6 characters, each one from the exact set A, b, 0, 1.

vector<char> <name> <length> <charset>;

            

            

              
Parameters

    • 
                
  • length — number of characters

  • charset — binary | lowercase | uppercase | digits | alpha | alphanum | custom{...} for an exact set

Example

pair<int,int> / pair<long long,long long>

            

              
Template

              
pair<int,int> <name> <count> <aMin> <aMax> <bMin> <bMax> [unique];

            

            

              
Parameters

              
    
                
  • name — variable name
    • 
                
  • count — number of pairs
    • 
                
  • aMin, aMax — range for the first element
    • 
                
  • bMin, bMax — range for the second element
    • 
                
  • true (optional) — all pairs unique
    • 
              

            

            

              
Example

              
pair<int,int> p 5 1 10 20 30;

              
→ 5 pairs: first ∈ [1,10], second ∈ [20,30].

            

          


          

            
matrix<int> / matrix<long long>

            

              
Template

              
matrix<int> <name> <rows> <cols> <min> <max> [unique];

            

            

              
Parameters

              
    
                
  • name — variable name
    • 
                
  • rows — number of rows
    • 
                
  • cols — number of columns
    • 
                
  • min — minimum value per cell
    • 
                
  • max — maximum value per cell
    • 
                
  • true (optional) — unique values within each row
    • 
              

            

            

              
Example

              
matrix<int> m 3 4 0 9;

              
→ 3×4 matrix, each cell between 0 and 9.

            

          


          

            
graph

            

              
Template

              
graph <name> <nodes> <edges> <weighted> <directed> <simple>;

            

            

              
Parameters

              
    
                
  • name — variable name
    • 
                
  • nodes — number of vertices (1-indexed)
    • 
                
  • edges — number of edges
    • 
                
  • weighted — true to add random edge weights
    • 
                
  • directed — true for directed, false for undirected
    • 
                
  • simple — true to forbid self-loops and multi-edges
    • 
              

            

            

              
Example

              
graph g 5 6 true false true;

              
→ Undirected, weighted, simple graph with 5 nodes and 6 edges.

            

          


          

            
tree

            

              
Template

              
tree <name> <nodes> <weighted>;

            

            

              
Parameters

              
    
                
  • name — variable name
    • 
                
  • nodes — number of vertices
    • 
                
  • weighted — true to add random edge weights
    • 
              

            

            

              
Example

              
tree t 7 false;

              
→ Unweighted tree with 7 nodes.

            

          


          

            
perm

            

              
Template

              
perm <name> <size> <oneBased>;

            

            

              
Parameters

              
    
                
  • name — variable name
    • 
                
  • size — number of elements
    • 
                
  • oneBased — true for 1..n, false for 0..n−1
    • 
              

            

            

              
Example

              
perm p 8 true;

              
→ Random permutation of 1..8.
